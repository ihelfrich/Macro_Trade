\section*{Propositions}

\newtheorem{proposition}{Proposition}

\begin{proposition}[Bounds]
With rerouting allowed and $\kappa \in [0,1]$, for each region $i$,
$$
   \lambda_i^{1/\epsilon^\ast} \le W_i'/W_i \le 1,
$$
where $\epsilon^\ast$ is an $\alpha$-weighted harmonic mean of $\epsilon_k$. Equality at left under pure severance (no reroute), equality at right at $\kappa=0$.
\end{proposition}

\begin{proposition}[Convexity in concentration]
Let $s$ be the Herfindahl-style exposure to top cross-bloc links. Then $W'/W$ is convex decreasing in $s$ under $\zeta$ below a threshold tied to $\epsilon$.
\end{proposition}

\begin{proposition}[IO amplification]
If $A$ is nonnegative and spectral radius $\rho(A) < 1$, then aggregate losses weakly increase in $\rho(A)$. Strict if shocks hit upstream sectors with positive Domar weights.
\end{proposition}

\begin{proposition}[Distributional widening under non-homothetic demand]
The variance of household-level welfare losses rises with $\kappa$ when Engel curves tilt toward imported luxuries.
\end{proposition}
